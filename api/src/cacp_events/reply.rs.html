1//! The answer half of a request that arrived as an [`Event`](crate::Event).
2
3use std::fmt;
4use tokio::sync::oneshot;
5
6/// A request waiting on the consumer.
7///
8/// Dropping it declines the request with `method not found` — the same refusal
9/// [`Client`](cacp::Client) sends for a method left unimplemented.
10pub struct Reply<T>(pub(crate) oneshot::Sender<proto::Result<T>>);
11
12impl<T> Reply<T> {
13    pub fn send(self, value: T) {
14        let _ = self.0.send(Ok(value));
15    }
16
17    pub fn fail(self, error: proto::Error) {
18        let _ = self.0.send(Err(error));
19    }
20
21    /// True once the agent has cancelled the request; a prompt still on screen
22    /// for it should come down.
23    pub fn is_cancelled(&self) -> bool {
24        self.0.is_closed()
25    }
26
27    /// Resolves when the agent cancels the request.
28    pub async fn cancelled(&mut self) {
29        self.0.closed().await;
30    }
31}
32
33impl<T> fmt::Debug for Reply<T> {
34    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
35        f.debug_struct("Reply")
36            .field("cancelled", &self.is_cancelled())
37            .finish()
38    }
39}

1//! Drive an ACP agent from a client.
2//!
3//! Implement [`Client`] and hand it to [`spawn`] or one of the `connect_*`
4//! functions. You get back an [`AgentConn`] for calling the agent — opening
5//! sessions, sending prompts, cancelling turns.
6//!
7//! Every optional method of [`Client`] defaults to replying "method not found",
8//! so an agent asking for something you do not serve gets a clean refusal and
9//! carries on.
10//!
11//! A frontend with an event loop of its own can take `cacp-events` instead,
12//! which serves this trait over a channel.
13

15use serve::Serve;

28/// Connect `client` to an agent over a duplex stream. Use [`spawn`] instead
29/// when the agent is a subprocess.
30pub fn connect<S, C>(stream: S, client: Arc<C>) -> AgentConn
31where
32    S: AsyncRead + AsyncWrite + Send + 'static,
33    C: Client,
34{
35    AgentConn(Peer::duplex(stream, Arc::new(Serve(client))))
36}
37
38/// Connect `client` to an agent over a reader and a writer owned separately.
39pub fn connect_on<R, W, C>(reader: R, writer: W, client: Arc<C>) -> AgentConn
40where
41    R: AsyncBufRead + Unpin + Send + 'static,
42    W: AsyncWrite + Unpin + Send + 'static,
43    C: Client,
44{
45    AgentConn(Peer::new(reader, writer, Arc::new(Serve(client))))
46}

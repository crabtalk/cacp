1//! What a side of the connection answers when the other side calls.
2
3use std::future::Future;
4
5/// The inbound half of a connection.
6///
7/// Implemented for you by `cacp-server` and `cacp-client`; you should not need
8/// to write it by hand.
9pub trait Handler: Send + Sync + 'static {
10    /// Answer a request. Returning `Err` sends that error back — it never
11    /// tears down the connection.
12    fn request(
13        &self,
14        method: String,
15        params: serde_json::Value,
16    ) -> impl Future<Output = proto::Result<serde_json::Value>> + Send;
17
18    /// Handle a notification. Awaited inline, so notifications stay in wire
19    /// order; keep it quick.
20    fn notification(
21        &self,
22        method: String,
23        params: serde_json::Value,
24    ) -> impl Future<Output = ()> + Send;
25}
26
27/// Params that don't match the method's shape are the caller's error.
28pub(crate) fn decode<T: serde::de::DeserializeOwned>(
29    params: serde_json::Value,
30) -> proto::Result<T> {
31    serde_json::from_value(params).map_err(|e| proto::Error::invalid_params().data(e.to_string()))
32}
33
34/// A handler returning something unserializable is our bug.
35pub(crate) fn encode<T: serde::Serialize>(value: T) -> proto::Result<serde_json::Value> {
36    serde_json::to_value(value).map_err(|e| proto::Error::internal_error().data(e.to_string()))
37}

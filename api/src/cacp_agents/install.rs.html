15/// A local installation: the command to spawn and its arguments.
16#[derive(Debug, Clone, Serialize, Deserialize)]
17pub struct Installed {
18    pub id: String,
19    pub version: String,
20    pub command: String,
21    #[serde(default)]
22    pub args: Vec<String>,
23}
24
25impl Agent {
26    /// Install under `data_dir`, streaming installer output to `on_line`.
27    /// Replaces any previous install of the same agent.
28    pub fn install(&self, data_dir: &Path, on_line: impl FnMut(&str)) -> Result<Installed> {
29        let Distribution::Npm { package, args } = &self.distribution else {
30            bail!("{} cannot be installed yet", self.name);
31        };
32        let command = install_npm(&agent_dir(data_dir, &self.id), package, on_line)?;
33
34        let record = Installed {
35            id: self.id.clone(),
36            version: self.version.clone(),
37            command,
38            args: args.clone(),
39        };
40        std::fs::write(
41            record_file(data_dir, &self.id),
42            serde_json::to_string_pretty(&record)?,
43        )
44        .context("recording the installation")?;
45        Ok(record)
46    }
47}
48
49impl Installed {
50    /// The recorded installation for `id`, if the agent is installed and its
51    /// command still exists.
52    pub fn find(data_dir: &Path, id: &str) -> Option<Self> {
53        let body = std::fs::read_to_string(record_file(data_dir, id)).ok()?;
54        let record: Self = serde_json::from_str(&body).ok()?;
55        Path::new(&record.command).exists().then_some(record)
56    }
57
58    /// Delete an installed agent. Succeeds whether or not it was there.
59    pub fn remove(data_dir: &Path, id: &str) -> Result<()> {
60        let dir = agent_dir(data_dir, id);
61        if dir.exists() {
62            std::fs::remove_dir_all(&dir).with_context(|| format!("removing {}", dir.display()))?;
63        }
64        Ok(())
65    }
66}
67
68fn agent_dir(data_dir: &Path, id: &str) -> PathBuf {
69    data_dir.join("agents").join(id)
70}
71
72fn record_file(data_dir: &Path, id: &str) -> PathBuf {
73    agent_dir(data_dir, id).join("install.json")
74}

76/// The package name in a spec like `@scope/name@1.2.3` — the version separator
77/// is the last `@` that isn't the scope's leading one.
78pub fn package_name(spec: &str) -> &str {
79    match spec.rfind('@') {
80        Some(ix) if ix > 0 => &spec[..ix],
81        _ => spec,
82    }
83}

85/// Install one npm package into `dir` (replacing whatever was there), streaming
86/// installer output to `on_line`. Returns the executable path.
87pub(crate) fn install_npm(
88    dir: &Path,
89    package: &str,
90    mut on_line: impl FnMut(&str),
91) -> Result<String> {
92    if utils::which("npm").is_none() {
93        bail!("npm was not found on PATH — install Node.js to add agents");
94    }
95    let _ = std::fs::remove_dir_all(dir);
96    std::fs::create_dir_all(dir).with_context(|| format!("creating {}", dir.display()))?;
97
98    on_line(&format!("npm install {package}"));
99    let mut child = Command::new("npm")
100        .arg("install")
101        .arg("--prefix")
102        .arg(dir)
103        .args(["--no-fund", "--no-audit", "--loglevel", "http"])
104        .arg(package)
105        .stdin(Stdio::null())
106        .stdout(Stdio::piped())
107        .stderr(Stdio::piped())
108        .spawn()
109        .context("running npm")?;
110
111    // npm splits progress across both streams; merge them so the caller sees
112    // output in the order it arrives.
113    let (tx, rx) = std::sync::mpsc::channel::<String>();
114    for stream in [
115        child
116            .stdout
117            .take()
118            .map(|s| Box::new(s) as Box<dyn std::io::Read + Send>),
119        child
120            .stderr
121            .take()
122            .map(|s| Box::new(s) as Box<dyn std::io::Read + Send>),
123    ]
124    .into_iter()
125    .flatten()
126    {
127        let tx = tx.clone();
128        std::thread::spawn(move || {
129            for line in BufReader::new(stream).lines().map_while(Result::ok) {
130                if tx.send(line).is_err() {
131                    break;
132                }
133            }
134        });
135    }
136    drop(tx);
137    for line in rx {
138        let line = line.trim_end();
139        if !line.is_empty() {
140            on_line(line);
141        }
142    }
143
144    let status = child.wait().context("waiting for npm")?;
145    if !status.success() {
146        bail!("npm install failed ({status})");
147    }
148    let command = binary_path(dir, package)?;
149    on_line("installed");
150    Ok(command)
151}

153/// The executable npm linked for `package`, read from the installed package's
154/// own `bin` field.
155fn binary_path(dir: &Path, package: &str) -> Result<String> {
156    let name = package_name(package);
157    let manifest = dir.join("node_modules").join(name).join("package.json");
158    let body = std::fs::read_to_string(&manifest)
159        .with_context(|| format!("reading {}", manifest.display()))?;
160    let manifest: serde_json::Value = serde_json::from_str(&body)?;
161
162    let bin_name = match &manifest["bin"] {
163        serde_json::Value::String(_) => name.rsplit('/').next().unwrap_or(name).to_owned(),
164        serde_json::Value::Object(map) => map
165            .keys()
166            .next()
167            .cloned()
168            .ok_or_else(|| anyhow!("{name} declares no executable"))?,
169        _ => bail!("{name} declares no executable"),
170    };
171
172    let bin = dir.join("node_modules").join(".bin").join(&bin_name);
173    if !bin.exists() {
174        bail!("{} is missing after install", bin.display());
175    }
176    Ok(bin.display().to_string())
177}

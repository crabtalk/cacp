1//! The official MCP registry: search it, and install what it lists.
2//!
3//! Searching is server-side and live — the catalog runs to thousands of
4//! entries, so there is nothing to cache locally.
5
6use crate::install;
7use anyhow::{Context, Result, bail};
8use serde::Deserialize;
9use std::{
10    collections::HashSet,
11    path::{Path, PathBuf},
12};

14pub const SEARCH_URL: &str = "https://registry.modelcontextprotocol.io/v0/servers";
15
16/// How many results one search returns.
17const SEARCH_LIMIT: usize = 30;
18
19/// One MCP server from the registry.
20#[derive(Debug, Clone)]
21pub struct Server {
22    /// Reverse-DNS registry name, e.g. `io.github.owner/thing`.
23    pub id: String,
24    /// Short display name (the last path segment of `id`).
25    pub name: String,
26    pub description: String,
27    pub distribution: Distribution,
28}
29
30/// How to reach an MCP server.
31#[derive(Debug, Clone)]
32pub enum Distribution {
33    /// An npm package the agent runs over stdio.
34    Npm { package: String },
35    /// A remote server the agent connects to over HTTP.
36    Remote { url: String },
37    /// Python, containers, and other kinds that cannot be launched yet.
38    Unsupported { kind: String },
39}
40
41impl Server {
42    pub fn installable(&self) -> bool {
43        !matches!(self.distribution, Distribution::Unsupported { .. })
44    }
45
46    /// Remote servers need the agent to advertise `mcp_capabilities.http`.
47    pub fn is_remote(&self) -> bool {
48        matches!(self.distribution, Distribution::Remote { .. })
49    }
50
51    /// Make this server launchable: npm packages are installed and their
52    /// executable returned; remote servers need nothing, and give `None`.
53    pub fn install(&self, data_dir: &Path, on_line: impl FnMut(&str)) -> Result<Option<String>> {
54        match &self.distribution {
55            Distribution::Npm { package } => {
56                let dir = server_dir(data_dir, &self.id);
57                Ok(Some(install::install_npm(&dir, package, on_line)?))
58            }
59            Distribution::Remote { .. } => Ok(None),
60            Distribution::Unsupported { kind } => {
61                bail!(
62                    "{} is distributed as {kind}, which cannot be launched yet",
63                    self.name
64                )
65            }
66        }
67    }
68}

70#[derive(Deserialize)]
71struct WireResponse {
72    #[serde(default)]
73    servers: Vec<WireEntry>,
74}

76#[derive(Deserialize)]
77struct WireEntry {
78    server: WireServer,

81#[derive(Deserialize)]
82#[serde(rename_all = "camelCase")]
83struct WireServer {
84    name: String,
85    #[serde(default)]
86    title: Option<String>,
87    #[serde(default)]
88    description: String,
89    #[serde(default)]
90    packages: Vec<WirePackage>,
91    #[serde(default)]
92    remotes: Vec<WireRemote>,
93}
94
95#[derive(Deserialize)]
96#[serde(rename_all = "camelCase")]
97struct WirePackage {
98    #[serde(default)]
99    registry_type: String,
100    identifier: String,
101    #[serde(default)]
102    version: Option<String>,
103}
104
105#[derive(Deserialize)]
106#[serde(rename_all = "camelCase")]
107struct WireRemote {
108    #[serde(rename = "type", default)]
109    kind: String,
110    url: String,
111}

113fn convert(server: WireServer) -> Server {
114    // Prefer a package we can launch over stdio: every agent supports stdio,
115    // while HTTP depends on the agent's capabilities.
116    let npm = server
117        .packages
118        .iter()
119        .find(|p| p.registry_type == "npm")
120        .map(|p| Distribution::Npm {
121            package: match &p.version {
122                Some(version) if !version.is_empty() => {
123                    format!("{}@{}", p.identifier, version)
124                }
125                _ => p.identifier.clone(),
126            },
127        });
128    let remote = server
129        .remotes
130        .iter()
131        .find(|r| r.kind == "streamable-http")
132        .map(|r| Distribution::Remote { url: r.url.clone() });
133    let distribution = npm.or(remote).unwrap_or_else(|| Distribution::Unsupported {
134        kind: server
135            .packages
136            .first()
137            .map(|p| p.registry_type.clone())
138            .or_else(|| server.remotes.first().map(|r| r.kind.clone()))
139            .unwrap_or_else(|| "unknown".to_owned()),
140    });
141
142    let name = server.title.clone().unwrap_or_else(|| {
143        server
144            .name
145            .rsplit('/')
146            .next()
147            .unwrap_or(&server.name)
148            .to_owned()
149    });
150    Server {
151        id: server.name,
152        name,
153        description: server.description,
154        distribution,
155    }
156}

158/// Read a search response, for a caller that fetched it itself.
159pub fn parse(body: &str) -> Result<Vec<Server>> {
160    let response: WireResponse =
161        serde_json::from_str(body).context("malformed MCP registry response")?;
162    Ok(dedupe(
163        response.servers.into_iter().map(|e| convert(e.server)),
164    ))
165}

167/// Search the registry. An empty query returns the newest entries.
168///
169/// `version=latest` is essential: without it the registry returns every
170/// published version of every server, so one name appears many times.
171pub fn search(query: &str) -> Result<Vec<Server>> {
172    let mut request = ureq::get(SEARCH_URL)
173        .query("limit", SEARCH_LIMIT.to_string())
174        .query("version", "latest");
175    if !query.trim().is_empty() {
176        request = request.query("search", query.trim());
177    }
178    let body = request
179        .call()
180        .context("searching the MCP registry")?
181        .body_mut()
182        .read_to_string()
183        .context("reading MCP registry results")?;
184    parse(&body)
185}

187/// Keep the first entry per id — belt and braces behind `version=latest`.
188fn dedupe(servers: impl Iterator<Item = Server>) -> Vec<Server> {
189    let mut seen = HashSet::new();
190    servers.filter(|s| seen.insert(s.id.clone())).collect()
191}

193/// Where an installed MCP server lives.
194fn server_dir(data_dir: &Path, id: &str) -> PathBuf {
195    // Registry ids contain `/`, which would nest directories.
196    data_dir.join("mcp").join(id.replace('/', "_"))
197}

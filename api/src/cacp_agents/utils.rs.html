1//! Small helpers with no home of their own.

3use anyhow::{Result, bail};
4use std::path::{Component, Path, PathBuf};
5
6/// `Command` for `program` as [`which::which`] found it. A Windows program is
7/// run through the file its extension names — `npm.cmd` through `cmd.exe`,
8/// which the standard library arranges — and without a console window of its
9/// own, which a GUI process would otherwise be handed for every child it
10/// starts.
11pub(crate) fn command(program: &Path) -> std::process::Command {
12    #[cfg_attr(not(windows), allow(unused_mut))]
13    let mut command = std::process::Command::new(program);
14    #[cfg(windows)]
15    {
16        use std::os::windows::process::CommandExt as _;
17        const CREATE_NO_WINDOW: u32 = 0x0800_0000;
18        command.creation_flags(CREATE_NO_WINDOW);
19    }
20    command
21}

23/// `dir` joined with a relative path the registry supplied — at least one
24/// component, every one a plain name. An id or a `cmd` that climbs out, or an
25/// empty one naming `dir` itself, would aim an install — or the
26/// `remove_dir_all` that undoes one — somewhere it was never given.
27pub fn contained(dir: &Path, rel: &str) -> Result<PathBuf> {
28    let rel = Path::new(rel.trim_start_matches("./"));
29    let mut parts = rel.components().peekable();
30    if parts.peek().is_none() || !parts.all(|c| matches!(c, Component::Normal(_))) {
31        bail!("{} is not a path inside {}", rel.display(), dir.display());
32    }
33    Ok(dir.join(rel))
34}
